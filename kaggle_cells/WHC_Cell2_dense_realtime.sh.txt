elapsed_min(){ echo $(( ($(date +%s) - SCRIPT_START_TS) / 60 )); }
remaining_min(){ echo $(( MAX_WALL_MIN - $(elapsed_min) )); }
can_start(){ local r; r="$(remaining_min)"; if (( r <= GUARD_MIN )); then log "Time guard reached. Remaining=${r} min."; return 1; fi; return 0; }
run_cmd(){
  local name="$1"; shift; local out="$1"; shift; mkdir -p "$out"
  if [[ "$FORCE" != "1" && ( -f "$out/test_summary.json" || -f "$out/summary.json" ) ]]; then
    log "SKIP $name"; echo -e "$name\tSKIP\t0\t$out" >> "$OUT_ROOT/status.tsv"; return 0
  fi
  can_start || return 0
  log "START $name"; set +e; "$@" 2>&1 | tee "$out/stdout.log"; local code=${PIPESTATUS[0]}; set -e
  if [[ $code -ne 0 ]]; then log "FAILED $name with code $code"; echo -e "$name\tFAILED\t$code\t$out" >> "$OUT_ROOT/status.tsv"; return 0; fi
  log "DONE $name"; echo -e "$name\tDONE\t0\t$out" >> "$OUT_ROOT/status.tsv"
}

def metrics(conf):
    conf=conf.astype(float); tp=np.diag(conf); iou=tp/np.maximum(conf.sum(1)+conf.sum(0)-tp,1); dice=2*tp/np.maximum(conf.sum(1)+conf.sum(0),1); acc=tp/np.maximum(conf.sum(1),1)
    return dict(mIoU_percent=float(iou.mean()*100),dice_percent=float(dice.mean()*100),pixel_acc_percent=float(tp.sum()/max(conf.sum(),1)*100),mean_acc_percent=float(acc.mean()*100))

        if ev['mIoU_percent']>best: best=ev['mIoU_percent']; be=e; torch.save(m.state_dict(),out/'best.pth')
