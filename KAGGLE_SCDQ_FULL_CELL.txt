%%bash
set -Eeuo pipefail

# ============================================================
# SCDQ-DT1D v0.9.1: corrected Kaggle runner
# Main comparison: Table 14-15, 10 methods x 3 seeds = 30 runs
# SCDQ ablation: 13 variants x 3 seeds = 39 runs
# Dataset/backbone: Caltech101 + ResNet-18
# ============================================================

REPO_URL="${REPO_URL:-https://github.com/tydeptrai21042004/1D-DT-ADAPTER.git}"
BRANCH="${BRANCH:-dt1d-v9-scdq-math-latency}"
SOURCE_MODE="${SOURCE_MODE:-auto}"   # auto | dataset | git
ARCHIVE_BASENAME="${ARCHIVE_BASENAME:-DT1D_V9_SCDQ_KAGGLE_FINAL.zip}"

SEEDS="${SEEDS:-0,1,2}"
[[ "$SEEDS" == "0,1,2" ]] || { echo "This publication cell requires SEEDS=0,1,2." >&2; exit 2; }
DEVICE="${DEVICE:-cuda}"
RUN_TESTS="${RUN_TESTS:-1}"
RUN_COMPARISON="${RUN_COMPARISON:-1}"
RUN_ABLATION="${RUN_ABLATION:-1}"

WORKDIR="${WORKDIR:-/kaggle/working}"
REPO_DIR="${REPO_DIR:-$WORKDIR/1D-DT-ADAPTER}"
DATA_DIR="${DATA_DIR:-$WORKDIR/data}"
OUTPUT_ROOT="${OUTPUT_ROOT:-$WORKDIR/dt1d_v9_scdq_full_results}"
COMPARISON_ROOT="$OUTPUT_ROOT/comparison"
ABLATION_ROOT="$OUTPUT_ROOT/ablation"
RESULT_ZIP="${RESULT_ZIP:-$WORKDIR/dt1d_v9_scdq_full_results.zip}"

export OUTPUT_ROOT COMPARISON_ROOT ABLATION_ROOT RESULT_ZIP

log() {
  echo
  echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*"
}

package_results() {
  local code=$?
  trap - EXIT
  set +e
  if [[ -d "$OUTPUT_ROOT" ]]; then
    python - <<'PYZIP'
import os
import shutil
from pathlib import Path
root = Path(os.environ["OUTPUT_ROOT"])
zip_path = Path(os.environ["RESULT_ZIP"])
zip_path.parent.mkdir(parents=True, exist_ok=True)
if zip_path.exists():
    zip_path.unlink()
created = Path(shutil.make_archive(
    str(zip_path.with_suffix("")), "zip",
    root_dir=root.parent, base_dir=root.name,
))
if created != zip_path:
    created.replace(zip_path)
print(f"[package] Created: {zip_path}")
PYZIP
  fi
  exit "$code"
}
trap package_results EXIT

mkdir -p "$WORKDIR" "$DATA_DIR" "$OUTPUT_ROOT"
rm -rf "$REPO_DIR"

# ------------------------------------------------------------
# 1. Acquire source. Auto mode prefers a Kaggle Dataset archive,
#    then falls back to the GitHub branch.
# ------------------------------------------------------------
ARCHIVE=""
if [[ "$SOURCE_MODE" == "auto" || "$SOURCE_MODE" == "dataset" ]]; then
  ARCHIVE="$(find /kaggle/input -type f -name "$ARCHIVE_BASENAME" -print -quit 2>/dev/null || true)"
fi

if [[ -n "$ARCHIVE" ]]; then
  log "Using source archive: $ARCHIVE"
  EXTRACT_DIR="$WORKDIR/dt1d_v9_source_extract"
  rm -rf "$EXTRACT_DIR"
  mkdir -p "$EXTRACT_DIR"
  unzip -q "$ARCHIVE" -d "$EXTRACT_DIR"
  MAIN_FILE="$(find "$EXTRACT_DIR" -type f -name main.py -print -quit)"
  [[ -n "$MAIN_FILE" ]] || { echo "main.py was not found in $ARCHIVE" >&2; exit 2; }
  SOURCE_DIR="$(dirname "$MAIN_FILE")"
  cp -a "$SOURCE_DIR" "$REPO_DIR"
  printf 'archive=%s\n' "$ARCHIVE" > "$OUTPUT_ROOT/source_revision.txt"
elif [[ "$SOURCE_MODE" == "dataset" ]]; then
  echo "Dataset mode requested, but $ARCHIVE_BASENAME was not found under /kaggle/input." >&2
  exit 2
else
  log "Cloning GitHub branch: $BRANCH"
  if ! git clone --branch "$BRANCH" --single-branch "$REPO_URL" "$REPO_DIR"; then
    echo "Failed to clone branch '$BRANCH'. Push the provided repository first, or upload" >&2
    echo "$ARCHIVE_BASENAME as a Kaggle Dataset and keep SOURCE_MODE=auto." >&2
    exit 2
  fi
fi

cd "$REPO_DIR"
if [[ -d .git ]]; then
  [[ "$(git branch --show-current)" == "$BRANCH" ]] || {
    echo "Expected branch '$BRANCH', found '$(git branch --show-current)'." >&2
    exit 2
  }
  git rev-parse HEAD | tee "$OUTPUT_ROOT/git_commit.txt"
  git status --short --branch | tee "$OUTPUT_ROOT/git_status.txt"
fi

# ------------------------------------------------------------
# 2. Install only missing experiment add-ons. Preserve Kaggle's
#    existing CUDA/PyTorch/NumPy stack.
# ------------------------------------------------------------
log "Installing runtime add-ons"
python -m pip install -q --upgrade-strategy only-if-needed \
  "timm==1.0.15" \
  "thop==0.1.1.post2209072238" \
  "fvcore==0.1.5.post20221221" \
  "torchmetrics==1.6.1" \
  "pytest>=8,<9"

python - <<'PYDEPS'
import importlib.util
import subprocess
import sys
required = {
    "yaml": "PyYAML", "pandas": "pandas", "numpy": "numpy",
    "scipy": "scipy", "sklearn": "scikit-learn", "PIL": "Pillow",
}
missing = [pkg for mod, pkg in required.items() if importlib.util.find_spec(mod) is None]
if missing:
    subprocess.check_call([sys.executable, "-m", "pip", "install", "-q",
                           "--upgrade-strategy", "only-if-needed", *missing])
print("Missing scientific packages installed:" if missing else "Core scientific packages present.", missing)
PYDEPS

# ------------------------------------------------------------
# 3. Runtime and source audits.
# ------------------------------------------------------------
log "Checking CUDA and versions"
python - <<'PYCUDA'
import importlib.metadata as md
import torch, torchvision
if not torch.cuda.is_available():
    raise SystemExit("CUDA unavailable. Enable a Kaggle GPU accelerator.")
print("PyTorch:", torch.__version__)
print("torchvision:", torchvision.__version__)
print("CUDA runtime:", torch.version.cuda)
print("cuDNN:", torch.backends.cudnn.version())
print("GPU:", torch.cuda.get_device_name(0))
for pkg in ("timm", "numpy", "pandas", "scipy", "scikit-learn", "PyYAML"):
    try: print(f"{pkg}:", md.version(pkg))
    except md.PackageNotFoundError: print(f"{pkg}: not installed")
PYCUDA

log "Auditing executable assets"
python - <<'PYFILES'
from pathlib import Path
required = [
    "main.py", "engine.py", "models/dt1d_adapter.py",
    "configs/paper/cnn_three_seed_manifest.yaml",
    "configs/experiments/scdq_three_seed_manifest.yaml",
    "tools/run_from_config.py", "tools/run_cnn_paper.py",
    "tools/aggregate_cnn_paper.py", "tools/preflight_cnn_matrix.py",
    "tools/validate_mlq_theory.py",
    "splits/caltech101/seed0_holdout20.json",
    "splits/caltech101/seed1_holdout20.json",
    "splits/caltech101/seed2_holdout20.json",
]
missing = [p for p in required if not Path(p).is_file()]
if missing:
    raise SystemExit("Missing required files:\n" + "\n".join(missing))
print(f"Asset audit passed: {len(required)} files.")
PYFILES

if [[ "$RUN_TESTS" == "1" ]]; then
  log "Running source tests"
  python -m pytest -q | tee "$OUTPUT_ROOT/pytest.txt"
fi

log "Validating quotient rank, nullspace, equivalence, and stability"
python tools/validate_mlq_theory.py | tee "$OUTPUT_ROOT/theory_validation_stdout.txt"
cp outputs/mlq_validation/theory_validation.json "$OUTPUT_ROOT/theory_validation.json"

# ------------------------------------------------------------
# 4. Preflight both matrices without downloading datasets.
# ------------------------------------------------------------
log "Preflighting main Table 14-15 comparison"
python tools/preflight_cnn_matrix.py \
  --manifest configs/paper/cnn_three_seed_manifest.yaml \
  --target table_14_15 \
  --output "$OUTPUT_ROOT/comparison_preflight.json"

log "Preflighting SCDQ ablation matrix"
python tools/preflight_cnn_matrix.py \
  --manifest configs/experiments/scdq_three_seed_manifest.yaml \
  --target table_14_15_scdq_ablation \
  --output "$OUTPUT_ROOT/ablation_preflight.json"

# ------------------------------------------------------------
# 5. Main fair comparison: final SCDQ proposal + all baselines.
# ------------------------------------------------------------
if [[ "$RUN_COMPARISON" == "1" ]]; then
  log "Building main comparison plan"
  python tools/run_cnn_paper.py \
    --manifest configs/paper/cnn_three_seed_manifest.yaml \
    --target table_14_15 --seeds "$SEEDS" --methods target \
    --data-path "$DATA_DIR" --device "$DEVICE" \
    --output-root "$COMPARISON_ROOT" --plan-only

  python - <<'PYPLAN'
import json, os
from pathlib import Path
plan = json.loads((Path(os.environ["COMPARISON_ROOT"]) / "execution_plan.json").read_text())
expected = {"dt1d", "full", "linear", "conv_r4", "bam", "residual",
            "ssf", "lora_conv", "bitfit", "sidetune"}
methods = {Path(r["output_dir"]).parts[-2] for r in plan["runs"]}
if len(plan["runs"]) != 30:
    raise SystemExit(f"Expected 30 comparison runs, got {len(plan['runs'])}")
if methods != expected:
    raise SystemExit(f"Comparison methods mismatch: {sorted(methods)}")
if plan["seeds"] != [0, 1, 2]:
    raise SystemExit(f"Seed mismatch: {plan['seeds']}")
print("Main plan passed: 10 methods x 3 seeds.")
PYPLAN

  log "Running main comparison"
  python tools/run_cnn_paper.py \
    --manifest configs/paper/cnn_three_seed_manifest.yaml \
    --target table_14_15 --seeds "$SEEDS" --methods target \
    --data-path "$DATA_DIR" --device "$DEVICE" \
    --output-root "$COMPARISON_ROOT" --skip-if-complete

  python tools/aggregate_cnn_paper.py \
    --root "$COMPARISON_ROOT" --target table_14_15 --require-seeds "$SEEDS"
fi

# ------------------------------------------------------------
# 6. Thirteen-variant mathematical/architectural ablation.
# ------------------------------------------------------------
if [[ "$RUN_ABLATION" == "1" ]]; then
  log "Building SCDQ ablation plan"
  python tools/run_cnn_paper.py \
    --manifest configs/experiments/scdq_three_seed_manifest.yaml \
    --target table_14_15_scdq_ablation --seeds "$SEEDS" --methods target \
    --data-path "$DATA_DIR" --device "$DEVICE" \
    --output-root "$ABLATION_ROOT" --plan-only

  python - <<'PYABL'
import json, os
from pathlib import Path
plan = json.loads((Path(os.environ["ABLATION_ROOT"]) / "execution_plan.json").read_text())
variants = {Path(r["output_dir"]).parts[-2].split("dt1d__", 1)[-1] for r in plan["runs"]}
if len(plan["runs"]) != 39:
    raise SystemExit(f"Expected 39 ablation runs, got {len(plan['runs'])}")
if len(variants) != 13:
    raise SystemExit(f"Expected 13 ablation variants, got {len(variants)}: {sorted(variants)}")
print("Ablation plan passed: 13 variants x 3 seeds.")
PYABL

  log "Running SCDQ ablations"
  python tools/run_cnn_paper.py \
    --manifest configs/experiments/scdq_three_seed_manifest.yaml \
    --target table_14_15_scdq_ablation --seeds "$SEEDS" --methods target \
    --data-path "$DATA_DIR" --device "$DEVICE" \
    --output-root "$ABLATION_ROOT" --skip-if-complete

  python tools/aggregate_cnn_paper.py \
    --root "$ABLATION_ROOT" --target table_14_15_scdq_ablation --require-seeds "$SEEDS"
fi

# ------------------------------------------------------------
# 7. Reproducibility snapshot and result listing.
# ------------------------------------------------------------
log "Recording environment"
python -m pip freeze > "$OUTPUT_ROOT/pip_freeze.txt"
nvidia-smi > "$OUTPUT_ROOT/nvidia_smi.txt" || true

log "Important outputs"
find "$OUTPUT_ROOT" -type f \
  \( -name "execution_plan.json" -o -name "raw_runs.csv" \
     -o -name "mean_std_numeric.csv" -o -name "mean_std_pretty.csv" \
     -o -name "manuscript_compact.csv" -o -name "manuscript_compact.tex" \
     -o -name "seed_completeness.json" -o -name "aggregation_summary.json" \
     -o -name "*preflight.json" -o -name "theory_validation.json" \) \
  -print | sort

log "COMPLETED"
echo "Results: $OUTPUT_ROOT"
echo "ZIP:     $RESULT_ZIP"
